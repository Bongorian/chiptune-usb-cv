cmake_minimum_required(VERSION 3.20)
option(CHIPTUNE_HOST_TESTS "Build portable engine tests instead of firmware" OFF)
if(NOT CHIPTUNE_HOST_TESTS)
  set(PICO_BOARD chiptune_rp2040 CACHE STRING "Hardware board")
  list(APPEND PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/boards")
  if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
  endif()
  if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to Pico SDK 2.3.1 (see README)")
  endif()
  include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(chiptune C CXX ASM)
set(CMAKE_C_STANDARD 11)
add_library(chiptune_core src/engine.c src/studio.c src/midi.c src/store.c src/power.c src/cv.c src/sn_clock.c src/oled.c)
target_include_directories(chiptune_core PUBLIC src)
target_compile_options(chiptune_core PRIVATE -Wall -Wextra -Werror -Wshadow -Wconversion -Wno-sign-conversion)
target_link_libraries(chiptune_core PUBLIC m)
if(CHIPTUNE_HOST_TESTS)
  enable_testing()
  option(CHIPTUNE_SANITIZE "Enable ASan and UBSan" ON)
  if(CHIPTUNE_SANITIZE)
    target_compile_options(chiptune_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(chiptune_core PUBLIC -fsanitize=address,undefined)
  endif()
  add_executable(engine_tests tests/test_engine.c tests/legacy_ui.c)
  target_link_libraries(engine_tests chiptune_core)
  target_compile_options(engine_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME engine_tests COMMAND engine_tests)
  add_executable(roles_tests tests/test_roles.c tests/legacy_ui.c)
  target_link_libraries(roles_tests chiptune_core)
  target_compile_options(roles_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME roles_tests COMMAND roles_tests)
  add_executable(performance_tests tests/test_performance.c tests/legacy_ui.c)
  target_link_libraries(performance_tests chiptune_core)
  target_compile_options(performance_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME performance_tests COMMAND performance_tests)
  add_executable(studio_tests tests/test_studio.c src/ui.c)
  target_link_libraries(studio_tests chiptune_core)
  target_compile_options(studio_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME studio_tests COMMAND studio_tests)
  add_executable(instrument_tests tests/test_instrument.c src/ui.c)
  target_link_libraries(instrument_tests chiptune_core)
  target_compile_options(instrument_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME instrument_tests COMMAND instrument_tests)
  add_executable(layout_tests tests/test_layout.c src/ui.c)
  target_link_libraries(layout_tests chiptune_core)
  target_compile_options(layout_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME layout_tests COMMAND layout_tests)
  add_executable(noise_tests tests/test_noise.c)
  target_link_libraries(noise_tests chiptune_core)
  target_compile_options(noise_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME noise_tests COMMAND noise_tests)
  add_executable(display_tests tests/test_display.c)
  target_link_libraries(display_tests chiptune_core)
  target_compile_options(display_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME display_tests COMMAND display_tests)
  add_executable(ui_preview tests/ui_preview.c src/ui.c)
  target_link_libraries(ui_preview chiptune_core)
else()
  set(PICOTOOL_GIT_BRANCH 2041936441b48a3cc53ae3da9e805229fe8f4e18 CACHE STRING "Pinned picotool")
  pico_sdk_init()
  target_compile_options(chiptune_core PRIVATE -fstack-usage)
  set(CHIPTUNE_USB_VID "0xcafe" CACHE STRING "Development USB VID; supply an assigned VID for distribution")
  set(CHIPTUNE_USB_PID "0x4008" CACHE STRING "USB PID")
  add_executable(chiptune src/main.c src/usb_descriptors.c src/ui.c)
  pico_generate_pio_header(chiptune ${CMAKE_CURRENT_LIST_DIR}/pio/sn_clock.pio)
  target_include_directories(chiptune PRIVATE . src)
  set_target_properties(chiptune PROPERTIES LINKER_LANGUAGE C)
  target_compile_definitions(chiptune PRIVATE PICO_CXX_DISABLE_ALLOCATION_OVERRIDES=1 CHIPTUNE_USB_VID=${CHIPTUNE_USB_VID} CHIPTUNE_USB_PID=${CHIPTUNE_USB_PID})
  target_compile_options(chiptune PRIVATE -Wall -Wextra -Werror -fstack-usage)
  target_link_libraries(chiptune chiptune_core pico_stdlib pico_unique_id hardware_pwm hardware_adc hardware_i2c hardware_uart hardware_flash hardware_sync hardware_pio tinyusb_device tinyusb_board)
  # Two 32KiB banks hold complete scenes, sound library and calibration.
  pico_override_flash_size(chiptune 2031616)
  pico_enable_stdio_usb(chiptune 0)
  pico_enable_stdio_uart(chiptune 0)
  pico_add_extra_outputs(chiptune)
endif()
