cmake_minimum_required(VERSION 3.20)
option(CHIPTUNE_HOST_TESTS "Build portable engine tests instead of firmware" OFF)
if(NOT CHIPTUNE_HOST_TESTS)
  set(PICO_BOARD chiptune_rp2040 CACHE STRING "Hardware board")
  list(APPEND PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/boards")
  if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
  endif()
  if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to Pico SDK 2.3.1 (see README)")
  endif()
  include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(chiptune C CXX ASM)
set(CMAKE_C_STANDARD 11)
add_library(chiptune_core src/model.c src/resources.c src/engine.c src/render.c src/sequence.c src/register_bus.c src/sound_math.c src/chip_math.c src/midi.c src/store.c src/persistence.c src/power.c src/cv.c src/sn_clock.c src/oled.c src/editor.c)
target_include_directories(chiptune_core PUBLIC src)
target_compile_options(chiptune_core PRIVATE -Wall -Wextra -Werror -Wshadow)
target_link_libraries(chiptune_core PUBLIC m)
set(CHIPTUNE_SN1_TRIM "0.51627034f" CACHE STRING "Left SN nominal gain")
set(CHIPTUNE_SN2_TRIM "0.51627034f" CACHE STRING "Right SN nominal gain")
target_compile_definitions(chiptune_core PRIVATE CHIPTUNE_SN1_TRIM=${CHIPTUNE_SN1_TRIM} CHIPTUNE_SN2_TRIM=${CHIPTUNE_SN2_TRIM})
if(CHIPTUNE_HOST_TESTS)
  enable_testing()
  option(CHIPTUNE_SANITIZE "ASan and UBSan" ON)
  if(CHIPTUNE_SANITIZE)
    target_compile_options(chiptune_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(chiptune_core PUBLIC -fsanitize=address,undefined)
  endif()
  foreach(name synth storage bus protocol ui sequence midi display)
    add_executable(${name}_tests tests/test_${name}.c src/ui.c)
    target_link_libraries(${name}_tests chiptune_core)
    target_compile_options(${name}_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME ${name}_tests COMMAND ${name}_tests)
  endforeach()
  foreach(name editor_export pitch_export ui_preview editor_bridge performance)
    add_executable(${name} tools/${name}.c src/ui.c)
    target_link_libraries(${name} chiptune_core)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endforeach()
else()
  set(PICOTOOL_GIT_BRANCH 2041936441b48a3cc53ae3da9e805229fe8f4e18 CACHE STRING "Pinned picotool")
  pico_sdk_init()
  target_compile_options(chiptune_core PRIVATE -fstack-usage)
  set(CHIPTUNE_USB_VID "0xcafe" CACHE STRING "Development USB VID; supply an assigned VID for distribution")
  set(CHIPTUNE_USB_PID "0x4008" CACHE STRING "USB PID")
  add_executable(chiptune src/main.c src/usb_descriptors.c src/ui.c)
  pico_generate_pio_header(chiptune ${CMAKE_CURRENT_LIST_DIR}/pio/sn_clock.pio)
  target_include_directories(chiptune PRIVATE . src)
  set_target_properties(chiptune PROPERTIES LINKER_LANGUAGE C)
  target_compile_definitions(chiptune PRIVATE PICO_CXX_DISABLE_ALLOCATION_OVERRIDES=1 CHIPTUNE_USB_VID=${CHIPTUNE_USB_VID} CHIPTUNE_USB_PID=${CHIPTUNE_USB_PID})
  target_compile_options(chiptune PRIVATE -Wall -Wextra -Werror -fstack-usage)
  target_link_libraries(chiptune chiptune_core pico_stdlib pico_unique_id hardware_pwm hardware_adc hardware_i2c hardware_uart hardware_flash hardware_sync hardware_pio tinyusb_device tinyusb_board)
  # Reserve two 32KiB banks in the LINKER only. The SDK flash driver must retain
  # the physical 2MiB capacity, otherwise every settings write hard-asserts.
  set(PICO_FLASH_SIZE_BYTES_STRING 2031616)
  configure_file(${PICO_SDK_PATH}/src/rp2_common/pico_standard_link/pico_flash_region.template.ld
    ${CMAKE_CURRENT_BINARY_DIR}/chiptune/pico_flash_region.ld)
  pico_add_linker_script_override_path(chiptune ${CMAKE_CURRENT_BINARY_DIR}/chiptune FILES pico_flash_region.ld)
  pico_enable_stdio_usb(chiptune 0)
  pico_enable_stdio_uart(chiptune 0)
  pico_add_extra_outputs(chiptune)
endif()
